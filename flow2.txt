    Ext_TMX["External TMX API"]:::external
    Ext_SIRA["External SIRA (SOAP)"]:::external
    Ext_CRM["External CRM System"]:::external

    %% --- The Internal System ---
    subgraph "Fraud Check Ecosystem"
        direction TB

        %% Components
        API["Enquiry Service<br/>(API & DB Owner)"]:::internal
        DB[("Enquiry DB")]:::database
        
        Kafka{{"Kafka Event Bus"}}:::eventbus
        
        Proc["Enquiry Processor<br/>(Orchestrator)"]:::internal
        
        TMX_Adp["TMX Adapter<br/>(REST Wrapper)"]:::internal
        SIRA_Adp["SIRA Adapter<br/>(REST Wrapper)"]:::internal

    end

    %% --- Flows ---

    %% 1. Ingestion Flow
    Client -->|1. POST /enquire| API
    API <-->|Persist State| DB

    %% 2. Processing Flow
    Kafka -.->|3. Sub: EnquiryCreated| Proc

    %% 3. Decision Fan-Out
    Proc -.->|"6a. Pub: ENQUIRY_UPDATE<br/>(Cleared/Decline/Refer)"| Kafka
    Proc -.->|"6b. Pub: CASE_REFERRED<br/>(If Referred)"| Kafka

    %% 4. Handling Updates & Case Creation
    Kafka -.->|7. Sub: ENQUIRY_UPDATE| API
    Kafka -.->|8. Sub: CASE_REFERRED| Case_W
    
    Case_W -->|9. Create Case| Ext_CRM

    %% 5. Closing the Loop
    Ext_CRM -.->|"10. Webhook: Case Update"| API
